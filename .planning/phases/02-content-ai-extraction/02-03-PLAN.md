Create src/inngest/functions/process-pdf.ts:
```typescript
import { inngest } from '../client';
import { prisma } from '@/lib/prisma';
import { extractTextFromUrl, PagedText } from '@/lib/pdf/extract-text';
import { analyzePdfForOcr } from '@/lib/pdf/detect-scanned';
import { chunkTextWithCitations, Chunk } from '@/lib/pdf/chunk';

export const processPdf = inngest.createFunction(
  {
    id: 'process-pdf',
    retries: 3,
    onFailure: async ({ event, error }) => {
      // Update document status on failure
      const documentId = event.data.documentId;
      await prisma.document.update({
        where: { id: documentId },
        data: {
          status: 'failed',
          errorMessage: error.message,
        },
      });
    },
  },
  { event: 'pdf/uploaded' },
  async ({ event, step }) => {
    const { documentId, fileUrl, fileName } = event.data;

    // Step 1: Update status to processing
    await step.run('update-status-processing', async () => {
      await prisma.document.update({
        where: { id: documentId },
        data: { status: 'processing' },
      });
    });

    // Step 2: Extract text from PDF
    const textResult = await step.run('extract-text', async () => {
      const pagedText = await extractTextFromUrl(fileUrl);
      return pagedText;
    });

    // Step 3: Analyze for OCR needs
    const ocrAnalysis = await step.run('analyze-for-ocr', async () => {
      return analyzePdfForOcr(textResult);
    });

    // Step 4: Run OCR if needed (placeholder - full OCR requires pdf-to-image)
    let finalPages = textResult.pages;
    if (ocrAnalysis.isScanned) {
      // Note: Full OCR implementation requires rendering PDF pages to images
      // This step logs a warning for now - OCR will be enhanced when needed
      await step.run('log-ocr-needed', async () => {
        console.warn(
          `Document ${documentId} appears to be scanned (avg ${ocrAnalysis.avgCharsPerPage} chars/page). ` +
          `OCR rendering not yet implemented - using extracted text.`
        );

        // Update document with OCR warning
        await prisma.document.update({
          where: { id: documentId },
          data: {
            status: 'processing',
            errorMessage: 'Document may be scanned. Text extraction quality may be limited.',
          },
        });
      });
    }

    // Step 5: Update document with page count
    await step.run('update-page-count', async () => {
      await prisma.document.update({
        where: { id: documentId },
        data: { totalPages: textResult.totalPages },
      });
    });

    // Step 6: Chunk text with citations
    const chunks = await step.run('chunk-text', async () => {
      return chunkTextWithCitations(finalPages, documentId, {
        chunkSize: 500,
        overlap: 100,
      });
    });

    // Step 7: Store chunks in database
    const storedChunks = await step.run('store-chunks', async () => {
      // Delete any existing chunks (in case of retry)
      await prisma.sourceChunk.deleteMany({
        where: { documentId },
      });

      // Batch insert chunks
      const chunkData = chunks.map((chunk) => ({
        documentId,
        content: chunk.content,
        pageNumber: chunk.pageNumber,
        chunkIndex: chunk.chunkIndex,
        startChar: chunk.startChar,
        endChar: chunk.endChar,
      }));

      await prisma.sourceChunk.createMany({
        data: chunkData,
      });

      return chunkData.length;
    });

    // Step 8: Update status to extracting (ready for AI)
    await step.run('update-status-extracting', async () => {
      await prisma.document.update({
        where: { id: documentId },
        data: {
          status: 'extracting',
          errorMessage: null, // Clear any warnings
        },
      });
    });

    // Trigger next step in pipeline
    await step.sendEvent('trigger-extraction', {
      name: 'pdf/processing.complete',
      data: {
        documentId,
        chunksCreated: storedChunks,
        totalPages: textResult.totalPages,
      },
    });

    return {
      documentId,
      totalPages: textResult.totalPages,
      chunksCreated: storedChunks,
      ocrNeeded: ocrAnalysis.isScanned,
    };
  }
);
```
