'use server';

import { revalidatePath } from 'next/cache';
import { auth } from '@/lib/auth';
import prisma from '@/lib/prisma';
import { z } from 'zod';
import { questionOptionsSchema } from '@/lib/questions/validation';

const UpdateQuestionSchema = z.object({
  questionId: z.string().cuid(),
  questionText: z.string().min(1).max(2000),
  options: questionOptionsSchema.nullable().optional(),
  correctAnswer: z.string().max(1000),
  explanation: z.string().max(2000),
  sourceEvidence: z.string().max(2000).optional(),
});

export async function updateQuestion(formData: FormData) {
  const session = await auth();
  if (!session?.user?.id) {
    return { error: 'Unauthorized' };
  }

  const rawOptions = formData.get('options');
  let parsedOptions = null;
  if (rawOptions && rawOptions !== 'null') {
    try {
      parsedOptions = JSON.parse(rawOptions as string);
    } catch {
      return { error: 'Invalid options format' };
    }
  }

  const parsed = UpdateQuestionSchema.safeParse({
    questionId: formData.get('questionId'),
    questionText: formData.get('questionText'),
    options: parsedOptions,
    correctAnswer: formData.get('correctAnswer'),
    explanation: formData.get('explanation'),
    sourceEvidence: formData.get('sourceEvidence'),
  });

  if (!parsed.success) {
    return { error: 'Invalid input', details: parsed.error.flatten() };
  }

  // Verify user owns the document that contains this question
  const question = await prisma.curatedQuestion.findUnique({
    where: { id: parsed.data.questionId },
    include: {
      document: {
        select: { uploadedById: true },
      },
    },
  });

  if (!question || question.document.uploadedById !== session.user.id) {
    return { error: 'Question not found or access denied' };
  }

  // Update the question
  await prisma.curatedQuestion.update({
    where: { id: parsed.data.questionId },
    data: {
      questionText: parsed.data.questionText,
      options: parsed.data.options ?? undefined,
      correctAnswer: parsed.data.correctAnswer,
      explanation: parsed.data.explanation,
      sourceEvidence: parsed.data.sourceEvidence ?? question.sourceEvidence,
    },
  });

  revalidatePath('/documents');

  return { success: true };
}

export async function getQuestion(questionId: string) {
  const session = await auth();
  if (!session?.user?.id) {
    return null;
  }

  const question = await prisma.curatedQuestion.findUnique({
    where: { id: questionId },
    include: {
      document: {
        select: { uploadedById: true },
      },
    },
  });

  if (!question || question.document.uploadedById !== session.user.id) {
    return null;
  }

  return question;
}

export async function publishQuiz(quizId: string) {
  const session = await auth();
  if (!session?.user?.id) {
    return { error: 'Unauthorized' };
  }

  const quiz = await prisma.quiz.findUnique({
    where: { id: quizId },
    select: {
      id: true,
      createdById: true,
      teacherPreviewedAt: true,
      documentId: true,
    },
  });

  if (!quiz || quiz.createdById !== session.user.id) {
    return { error: 'Quiz not found' };
  }

  // CONT-06: Verify teacher has previewed the quiz
  if (!quiz.teacherPreviewedAt) {
    return { error: 'You must preview the quiz before publishing' };
  }

  await prisma.quiz.update({
    where: { id: quizId },
    data: {
      status: 'published',
      publishedAt: new Date(),
    },
  });

  revalidatePath(`/documents/${quiz.documentId}/quiz`);

  return { success: true };
}

export async function updateQuizSettings(formData: FormData) {
  const session = await auth();
  if (!session?.user?.id) {
    return { error: 'Unauthorized' };
  }

  const quizId = formData.get('quizId') as string;
  const title = formData.get('title') as string;
  const description = formData.get('description') as string;
  const timeLimit = formData.get('timeLimit');
  const shuffleQuestions = formData.get('shuffleQuestions') === 'true';

  const quiz = await prisma.quiz.findUnique({
    where: { id: quizId },
    select: { createdById: true, documentId: true },
  });

  if (!quiz || quiz.createdById !== session.user.id) {
    return { error: 'Quiz not found' };
  }

  await prisma.quiz.update({
    where: { id: quizId },
    data: {
      title,
      description: description || null,
      timeLimit: timeLimit ? parseInt(timeLimit as string) : null,
      shuffleQuestions,
    },
  });

  revalidatePath(`/documents/${quiz.documentId}/quiz`);

  return { success: true };
}

'use client';

import { useState } from 'react';
import { useRouter } from 'next/navigation';
import { updateQuestion } from '@/actions/questions';
import type { CuratedQuestion } from '@/generated/prisma/client';
import type { QuestionOptions } from '@/lib/questions/types';

interface QuestionEditorProps {
  question: CuratedQuestion;
  onClose: () => void;
}

export function QuestionEditor({ question, onClose }: QuestionEditorProps) {
  const router = useRouter();
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const [questionText, setQuestionText] = useState(question.questionText);
  const [correctAnswer, setCorrectAnswer] = useState(question.correctAnswer);
  const [explanation, setExplanation] = useState(question.explanation);
  const [sourceEvidence, setSourceEvidence] = useState(question.sourceEvidence);
  const [options, setOptions] = useState<QuestionOptions | null>(
    question.options as QuestionOptions | null
  );

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setSaving(true);
    setError(null);

    const formData = new FormData();
    formData.set('questionId', question.id);
    formData.set('questionText', questionText);
    formData.set('correctAnswer', correctAnswer);
    formData.set('explanation', explanation);
    formData.set('sourceEvidence', sourceEvidence);
    formData.set('options', JSON.stringify(options));

    const result = await updateQuestion(formData);

    if (result.error) {
      setError(result.error);
      setSaving(false);
    } else {
      router.refresh();
      onClose();
    }
  };

  // Render type-specific option editors
  const renderOptionsEditor = () => {
    if (!options) return null;

    switch (options.type) {
      case 'multiple_choice':
        return (
          <div className="space-y-2">
            <label className="block text-sm font-medium">Answer Choices</label>
            {options.choices.map((choice, idx) => (
              <div key={choice.id} className="flex items-center gap-2">
                <input
                  type="radio"
                  checked={choice.isCorrect}
                  onChange={() => {
                    const newChoices = options.choices.map((c, i) => ({
                      ...c,
                      isCorrect: i === idx,
                    }));
                    setOptions({ ...options, choices: newChoices });
                    setCorrectAnswer(choice.id);
                  }}
                  className="h-4 w-4"
                />
                <span className="font-medium">{choice.id}.</span>
                <input
                  type="text"
                  value={choice.text}
                  onChange={(e) => {
                    const newChoices = [...options.choices];
                    newChoices[idx] = { ...newChoices[idx], text: e.target.value };
                    setOptions({ ...options, choices: newChoices });
                  }}
                  className="flex-1 rounded border px-2 py-1"
                />
              </div>
            ))}
          </div>
        );

      case 'true_false':
        return (
          <div>
            <label className="block text-sm font-medium">Correct Answer</label>
            <select
              value={options.correctAnswer ? 'true' : 'false'}
              onChange={(e) => {
                const val = e.target.value === 'true';
                setOptions({ ...options, correctAnswer: val });
                setCorrectAnswer(val.toString());
              }}
              className="mt-1 rounded border px-3 py-2"
            >
              <option value="true">True</option>
              <option value="false">False</option>
            </select>
          </div>
        );

      // Add more type editors as needed...
      default:
        return (
          <div className="rounded bg-gray-100 p-3 text-sm text-gray-600">
            Options editing for {options.type} not yet implemented.
            Raw JSON can be edited in database.
          </div>
        );
    }
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50">
      <div className="max-h-[90vh] w-full max-w-2xl overflow-y-auto rounded-lg bg-white p-6">
        <div className="mb-4 flex items-center justify-between">
          <h2 className="text-xl font-bold">Edit Question</h2>
          <button
            onClick={onClose}
            className="text-gray-500 hover:text-gray-700"
          >
            Close
          </button>
        </div>

        {error && (
          <div className="mb-4 rounded bg-red-100 p-3 text-red-700">{error}</div>
        )}

        <form onSubmit={handleSubmit} className="space-y-4">
          <div>
            <label className="block text-sm font-medium">Question Text</label>
            <textarea
              value={questionText}
              onChange={(e) => setQuestionText(e.target.value)}
              rows={3}
              className="mt-1 w-full rounded border px-3 py-2"
            />
          </div>

          {renderOptionsEditor()}

          <div>
            <label className="block text-sm font-medium">Explanation</label>
            <textarea
              value={explanation}
              onChange={(e) => setExplanation(e.target.value)}
              rows={2}
              className="mt-1 w-full rounded border px-3 py-2"
            />
          </div>

          <div>
            <label className="block text-sm font-medium">Source Evidence</label>
            <textarea
              value={sourceEvidence}
              onChange={(e) => setSourceEvidence(e.target.value)}
              rows={2}
              className="mt-1 w-full rounded border px-3 py-2"
            />
          </div>

          <div className="flex justify-end gap-3 pt-4">
            <button
              type="button"
              onClick={onClose}
              className="rounded border px-4 py-2 hover:bg-gray-50"
            >
              Cancel
            </button>
            <button
              type="submit"
              disabled={saving}
              className="rounded bg-blue-600 px-4 py-2 text-white hover:bg-blue-700 disabled:bg-blue-300"
            >
              {saving ? 'Saving...' : 'Save Changes'}
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}

import Link from 'next/link';
import { redirect } from 'next/navigation';
import { auth } from '@/lib/auth';
import { getQuizWithQuestions, publishQuiz } from '@/actions/quiz';
import { QuizDetailClient } from './quiz-detail-client';

export default async function QuizDetailPage({
  params,
}: {
  params: Promise<{ id: string; quizId: string }>;
}) {
  const { id: documentId, quizId } = await params;
  const session = await auth();

  if (!session?.user?.id) {
    redirect('/login');
  }

  const quiz = await getQuizWithQuestions(quizId);

  if (!quiz) {
    redirect(`/documents/${documentId}/quiz`);
  }

  const canPublish = quiz.teacherPreviewedAt && quiz.status !== 'published';
  const needsPreview = !quiz.teacherPreviewedAt;

  return (
    <div className="container mx-auto max-w-4xl p-6">
      {/* Header */}
      <div className="mb-6 flex items-start justify-between">
        <div>
          <h1 className="text-2xl font-bold">{quiz.title}</h1>
          {quiz.description && (
            <p className="mt-1 text-gray-600">{quiz.description}</p>
          )}
          <p className="mt-2 text-sm text-gray-500">
            {quiz.questions.length} questions
            {quiz.timeLimit && ` / ${quiz.timeLimit} min`}
          </p>
        </div>

        <div className="flex items-center gap-3">
          <span
            className={`rounded px-3 py-1 text-sm font-medium ${
              quiz.status === 'published'
                ? 'bg-green-100 text-green-800'
                : quiz.status === 'preview_required'
                  ? 'bg-yellow-100 text-yellow-800'
                  : 'bg-gray-100 text-gray-800'
            }`}
          >
            {quiz.status.replace('_', ' ')}
          </span>
        </div>
      </div>

      {/* Actions */}
      <div className="mb-6 flex flex-wrap gap-3">
        {needsPreview && (
          <Link
            href={`/documents/${documentId}/quiz/${quizId}/preview`}
            className="rounded bg-blue-600 px-4 py-2 text-white hover:bg-blue-700"
          >
            Preview Quiz (Required)
          </Link>
        )}

        {canPublish && (
          <form
            action={async () => {
              'use server';
              await publishQuiz(quizId);
            }}
          >
            <button
              type="submit"
              className="rounded bg-green-600 px-4 py-2 text-white hover:bg-green-700"
            >
              Publish Quiz
            </button>
          </form>
        )}

        <Link
          href={`/documents/${documentId}/quiz/${quizId}/edit`}
          className="rounded border border-gray-300 px-4 py-2 hover:bg-gray-50"
        >
          Edit Questions
        </Link>
      </div>

      {/* Questions list */}
      <QuizDetailClient
        documentId={documentId}
        quizId={quizId}
        questions={quiz.questions.map((qq) => qq.question)}
      />
    </div>
  );
}

'use client';

import { useState } from 'react';
import { QuestionEditor } from '@/components/questions/question-editor';
import type { CuratedQuestion } from '@/generated/prisma/client';

interface QuizDetailClientProps {
  documentId: string;
  quizId: string;
  questions: CuratedQuestion[];
}

export function QuizDetailClient({ questions }: QuizDetailClientProps) {
  const [editingQuestion, setEditingQuestion] = useState<CuratedQuestion | null>(null);

  return (
    <div>
      <h2 className="mb-4 text-lg font-medium">Questions</h2>

      <div className="space-y-4">
        {questions.map((question, idx) => (
          <div
            key={question.id}
            className="rounded-lg border border-gray-200 p-4"
          >
            <div className="flex items-start justify-between">
              <div className="flex-1">
                <div className="mb-2 flex items-center gap-2">
                  <span className="font-medium text-gray-500">Q{idx + 1}</span>
                  <span className="rounded bg-gray-100 px-2 py-0.5 text-xs">
                    {question.questionType.replace('_', ' ')}
                  </span>
                  <span className="rounded bg-purple-100 px-2 py-0.5 text-xs text-purple-700">
                    {question.bloomLevel}
                  </span>
                </div>
                <p className="text-gray-800">{question.questionText}</p>
                <p className="mt-2 text-sm text-gray-600">
                  <strong>Answer:</strong> {question.correctAnswer}
                </p>
              </div>
              <button
                onClick={() => setEditingQuestion(question)}
                className="ml-4 rounded border border-gray-300 px-3 py-1 text-sm hover:bg-gray-50"
              >
                Edit
              </button>
            </div>
          </div>
        ))}
      </div>

      {editingQuestion && (
        <QuestionEditor
          question={editingQuestion}
          onClose={() => setEditingQuestion(null)}
        />
      )}
    </div>
  );
}
